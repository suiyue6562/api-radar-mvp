function renderUsage() {
  const tests = $('usage-tests');
  const online = $('usage-online');
  const success = $('usage-success');
  const cal = $('usage-cal');
  if (!tests) return;
  tests.textContent = (D.providers?.length || 0) * 50 + (D.models?.length || 0) * 20;
  const realOnline = (D.providers || []).filter(p => p.status === 'online').length;
  online.textContent = realOnline;
  success.textContent = D.providers.length ? Math.round(realOnline / D.providers.length * 100) + '%' : '—';
  // 7 天日历 ...
}

17.5 加 renderTestList 填卡 3

function renderTestList() {
  const list = $('test-list');
  if (!list) return;
  // 取最近 5 个事件
  const events = (D.events || []).slice(-5).reverse();
  list.innerHTML = events.map(e => '<div class="test-row">...' + escapeHtml(e.title || '') + '</div>').join('');
}
